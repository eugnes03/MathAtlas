\begin{definition}[Open Set in a Metric Space]
  \label{def:open-set}
  Let $(X, \operatorname{dist})$ be a metric space (\ref{def:metricspaces}).
  A subset $U \subseteq X$ is \emph{open} if for every $x \in U$ there
  exists $\varepsilon > 0$ such that the open ball
  $$ B(x,\varepsilon) = \{ y \in X : \operatorname{dist}(x,y) < \varepsilon \} $$
  is contained in $U$.
\end{definition}
